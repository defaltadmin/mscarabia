// ============================================ // Animated Mesh Gradient + Particle Background // ============================================ var canvas = document.getElementById('hero-canvas'); if (canvas) { var ctx = canvas.getContext('2d'); var w, h, raf; var startTime = null; var particles = []; var PARTICLE_COUNT = window.innerWidth < 768 ? 30 : 60; function resizeCanvas() { var dpr = Math.min(window.devicePixelRatio || 1, 2); w = canvas.offsetWidth; h = canvas.offsetHeight; canvas.width = w * dpr; canvas.height = h * dpr; ctx.setTransform(dpr, 0, 0, dpr, 0, 0); } function Particle() { this.reset(); } Particle.prototype.reset = function() { this.x = Math.random() * w; this.y = Math.random() * h; this.vx = (Math.random() - 0.5) * 0.3; this.vy = -Math.random() * 0.5 - 0.1; this.life = 0; this.maxLife = 300 + Math.random() * 400; this.size = 1 + Math.random() * 2; var colors = ['230,57,70','74,158,255','139,92,246','6,182,212','240,192,64']; this.color = colors[Math.floor(Math.random() * colors.length)]; }; Particle.prototype.update = function() { this.x += this.vx; this.y += this.vy; this.vx += (Math.random() - 0.5) * 0.02; this.life++; if (this.life > this.maxLife || this.y < -10 || this.x < -10 || this.x > w + 10) this.reset(); }; Particle.prototype.draw = function() { var alpha = Math.sin((this.life / this.maxLife) * Math.PI) * 0.5; if (alpha < 0.01) return; ctx.beginPath(); ctx.arc(this.x, this.y, this.size, 0, Math.PI * 2); ctx.fillStyle = 'rgba(' + this.color + ',' + alpha + ')'; ctx.fill(); }; // Grid mesh function drawGrid() { var gridSize = 80; var time = (Date.now() - startTime) * 0.0001; ctx.lineWidth = 0.3; // Horizontal for (var y = 0; y < h; y += gridSize) { ctx.beginPath(); ctx.strokeStyle = 'rgba(230,57,70,' + (0.02 + Math.sin(y * 0.01 + time) * 0.01) + ')'; x < w; x += gridSize) { ctx.beginPath(); ctx.strokeStyle = 'rgba(74,158,255,' + (0.015 + Math.cos(x * 0.01 + time) * 0.008) + ')'; for (var y = 0; y < h; y += 4) { var offset = Math.cos(y * 0.005 + time + x * 0.01) * 2; if (y === 0) ctx.moveTo(x + offset, y); else ctx.lineTo(x + offset, y); } ctx.stroke(); } } // Glow orbs function drawOrbs(t) { var orbs = [ { x: w * 0.6 + Math.sin(t * 0.0003) * 80, y: h * 0.3 + Math.cos(t * 0.0004) * 60, r: Math.min(w * 0.3, 300), color: 'rgba(230,57,70,0.06)' }, { x: w * 0.2 + Math.cos(t * 0.0002) * 60, y: h * 0.7 + Math.sin(t * 0.0003) * 50, r: Math.min(w * 0.25, 250), color: 'rgba(74,158,255,0.04)' }, { x: w * 0.8 + Math.sin(t * 0.00025) * 70, y: h * 0.6 + Math.cos(t * 0.00035) * 40, r: Math.min(w * 0.2, 200), color: 'rgba(139,92,246,0.04)' } ]; ctx.globalCompositeOperation = 'lighter'; orbs.forEach(function(o) { var g = ctx.createRadialGradient(o.x, o.y, 0, o.x, o.y, o.r); g.addColorStop(0, o.color); g.addColorStop(1, 'transparent'); ctx.fillStyle = g; // Deep gradient base var bg = ctx.createRadialGradient(w * 0.5, h * 0.4, 0, w * 0.5, h * 0.4, Math.max(w, h) * 0.8); bg.addColorStop(0, '#0d0f18'); bg.addColorStop(0.5, '#080a12'); bg.addColorStop(1, '#04050a'); ctx.fillStyle = bg; ctx.fillRect(0, 0, w, h); drawGrid(); drawOrbs(t); // Particles for (var i = 0; i < particles.length; i++) { particles[i].update(); particles[i].draw(); } // Subtle horizontal scan line var scanY = (t * 0.05) % h; ctx.fillStyle = 'rgba(230,57,70,0.015)'; ctx.fillRect(0, scanY, w, 1); raf = requestAnimationFrame(drawBg); } resizeCanvas(); for (var i = 0; i < PARTICLE_COUNT; i++) particles.push(new Particle()); drawBg(0); window.addEventListener('resize', function() { resizeCanvas(); }); document.addEventListener('visibilitychange', function() { if (document.hidden) { cancelAnimationFrame(raf); startTime = null; } else { raf = requestAnimationFrame(drawBg);
